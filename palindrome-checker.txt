const inputBox = document.getElementById("text-input");
const checkBtn = document.getElementById("check-btn");
const results = document.getElementById("result");

function isPalindrome(str) {
  const regex = /[^A-Za-z0-9]/g;
  const makeLowerCase = str.toLowerCase().replace(regex, "");
  const reverseString = makeLowerCase.split("").reverse().join("");
  return makeLowerCase === reverseString;
};

function checkPalindrome() {
  const inputText = inputBox.value;
  const result = isPalindrome(inputText);
  if (!inputText) {
    results.innerText = "Please enter text to check";
  } else {
    results.innerText = result ? `${inputText} is a palindrome!` : `${inputText} is not a palindrome!`;
  };
};

checkBtn.addEventListener("click", checkPalindrome);
inputBox.addEventListener("keydown", function (event) {
  if (event.key === "Enter") {
    event.preventDefault();
    checkBtn.click();
  }
});
